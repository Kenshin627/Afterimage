cmake_minimum_required(VERSION 3.21)
project(Afterimage VERSION 1.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(CTest)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Network Concurrent)
if(MSVC)
    add_compile_options(/utf-8 /W4 /permissive-)
endif()
add_library(signal_core STATIC src/style.cpp src/renderer.cpp src/community.cpp src/animatedgif.cpp src/patina.cpp src/workshop.cpp src/workshopspatial.cpp src/creativeeffects.cpp src/creativemotion.cpp)
target_include_directories(signal_core PUBLIC src)
target_link_libraries(signal_core PUBLIC Qt6::Gui Qt6::Network)
add_library(studio_ui OBJECT src/editor.cpp src/files.cpp src/widgets.cpp src/inspector.cpp src/dialogs.cpp src/patinadialog.cpp src/workshopcontrols.cpp src/maskeditor.cpp)
target_link_libraries(studio_ui PUBLIC signal_core Qt6::Widgets Qt6::Concurrent)
set(app_sources src/main.cpp)
if(WIN32)
    list(APPEND app_sources assets/app.rc)
endif()
qt_add_executable(Afterimage WIN32 ${app_sources})
target_link_libraries(Afterimage PRIVATE studio_ui)
if(WIN32)
    target_compile_definitions(studio_ui PRIVATE NOMINMAX)
    target_link_libraries(studio_ui PUBLIC dwmapi)
endif()
qt_add_resources(studio_ui contents PREFIX "/" BASE assets FILES
    assets/coast.jpg assets/night.jpg assets/pool.jpg assets/watermark.png assets/app.png assets/theme.qss
    assets/BarlowCondensed-SemiBold.ttf assets/factory.json assets/community.json assets/gallery.json)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(signal_tests tests/signal_tests.cpp)
    target_compile_definitions(signal_tests PRIVATE REFERENCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/reference")
    target_link_libraries(signal_tests PRIVATE signal_core Qt6::Widgets Qt6::Test)
    add_test(NAME signal_tests COMMAND signal_tests)
    add_executable(editor_tests tests/editor_tests.cpp)
    target_compile_definitions(editor_tests PRIVATE REFERENCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/reference")
    target_link_libraries(editor_tests PRIVATE studio_ui Qt6::Test)
    add_test(NAME editor_tests COMMAND editor_tests)
    add_executable(patina_tests tests/patina_tests.cpp)
    target_compile_definitions(patina_tests PRIVATE REFERENCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/reference")
    target_link_libraries(patina_tests PRIVATE signal_core Qt6::Widgets Qt6::Test)
    add_test(NAME patina_tests COMMAND patina_tests)
    add_executable(workshop_tests tests/workshop_tests.cpp)
    target_compile_definitions(workshop_tests PRIVATE REFERENCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/reference")
    target_link_libraries(workshop_tests PRIVATE signal_core Qt6::Widgets Qt6::Test)
    add_test(NAME workshop_tests COMMAND workshop_tests)
endif()
